    // 全局状态变量
    let currentQuestionIndex = 0;
    let userAnswers = []; // 存储用户的选项tag

    // 页面切换核心函数
    function showPage(pageId) {
        document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
        document.getElementById(pageId).classList.add('active');
        window.scrollTo(0, 0);
    }

    // 开始测试
    function startQuiz() {
        currentQuestionIndex = 0;
        userAnswers = [];
        showPage('page-quiz');
        renderQuestion();
    }

    // 渲染题目
    function renderQuestion() {
        const q = questions[currentQuestionIndex];
        const total = questions.length;
        
        document.getElementById('question-counter').innerText = `问题 ${currentQuestionIndex + 1}/${total}`;
        document.getElementById('progress-bar').style.width = `${((currentQuestionIndex + 1) / total) * 100}%`;
        document.getElementById('question-title').innerText = q.title;

        const optionsList = document.getElementById('options-list');
        optionsList.innerHTML = '';

        q.options.forEach((opt, idx) => {
            const item = document.createElement('div');
            item.className = 'option-item';
            item.innerHTML = `<span>${String.fromCharCode(65 + idx)}. ${opt.text}</span>`;
            item.onclick = () => selectOption(opt.scoreTag, item);
            optionsList.appendChild(item);
        });
    }

    // 选择选项并进入下一题
    function selectOption(tag, element) {
        // 视觉高亮
        document.querySelectorAll('.option-item').forEach(el => el.classList.remove('selected'));
        element.classList.add('selected');

        userAnswers.push(tag);

        setTimeout(() => {
            currentQuestionIndex++;
            if (currentQuestionIndex < questions.length) {
                renderQuestion();
            } else {
                calculateResult();
            }
        }, 250); // 极短延迟提升交互手感
    }

    // 计算结果并展示最契合的城市
    function calculateResult() {
        // 统计用户选择的标签频次
        const tagCounts = {};
        userAnswers.forEach(tag => {
            tagCounts[tag] = (tagCounts[tag] || 0) + 1;
        });

        // 匹配算法：计算每个城市与用户选择的契合度得分
        let bestCity = citiesDatabase[0];
        let maxScore = -1;

        citiesDatabase.forEach(city => {
            let score = 0;
            city.tags.forEach(cityTag => {
                if (tagCounts[cityTag]) {
                    score += tagCounts[cityTag];
                }
            });
            // 加上一点随机扰动，防止每次相同选择命中同一固定城市
            score += Math.random() * 0.5;

            if (score > maxScore) {
                maxScore = score;
                bestCity = city;
            }
        });

        // 填充结果页数据
        document.getElementById('res-city').innerText = bestCity.name;
        document.getElementById('res-country').innerText = `? 属于你的本命栖息地 · ${bestCity.country}`;
        document.getElementById('res-copy').innerText = bestCity.copy;

        showPage('page-result');
    }

    // 重新测试
    function restartQuiz() {
        startQuiz();
    }
